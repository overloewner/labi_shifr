### Функция `is_prime(n)` — Проверка числа на простоту

```python
def is_prime(n):
    if n < 2:                          # Числа меньше 2 по определению не являются простыми
        return False
    for i in range(2, int(n**0.5) + 1):  # Проверяем делители от 2 до √n включительно.
                                           # Достаточно проверять до √n: если у n есть делитель
                                           # больше √n, то есть и соответствующий делитель меньше √n
        if n % i == 0:                 # Если n делится на i без остатка — n составное
            return False
    return True                        # Если делителей не найдено — n простое
```

---

### Функция `generate_prime()` — Генерация случайного простого числа

```python
def generate_prime(min_val=100, max_val=500):   # Диапазон по умолчанию: [100, 500]
    p = random.randint(min_val, max_val)         # Берём случайное целое число из диапазона
    while not is_prime(p):                       # Повторяем, пока число не окажется простым
        p = random.randint(min_val, max_val)     # Каждый раз берём новое случайное число
    return p                                     # Возвращаем найденное простое число
```

### Функция `mod_inverse(e, phi)` — Обратный элемент по модулю (расширенный алгоритм Евклида)

Цель: найти `d` такое, что `e * d ≡ 1 (mod phi)`.

```python
def mod_inverse(e, phi):
    old_r, r = e, phi    # old_r — текущий остаток, r — следующий остаток.
                          # Стартуем с самих чисел e и phi
    old_s, s = 1, 0      # Коэффициенты Безу: old_s отслеживает вклад e в НОД.
                          # В итоге old_s станет искомым обратным элементом
    while r != 0:                        # Продолжаем, пока остаток не обнулится
        q = old_r // r                   # Вычисляем частное от деления
        old_r, r = r, old_r - q * r     # Обновляем остатки (шаг алгоритма Евклида)
        old_s, s = s, old_s - q * s     # Обновляем коэффициенты Безу
    return old_s % phi   # Берём остаток по модулю phi, чтобы результат был положительным
```

### Функция `generate_keys()` — Генерация открытого и закрытого ключей RSA

```python
def generate_keys():
    p = generate_prime()       # Генерируем первое простое число p
    q = generate_prime()       # Генерируем второе простое число q
    while q == p:              # Условие: p и q должны быть различными
        q = generate_prime()   # Если совпали — генерируем q заново

    n = p * q                  # Модуль RSA: произведение двух простых чисел.
                                # n входит в оба ключа (открытый и закрытый)
    phi = (p - 1) * (q - 1)   # Функция Эйлера φ(n): количество чисел от 1 до n,
                                # взаимно простых с n. Для произведения двух простых
                                # φ(p*q) = (p-1)*(q-1)

    e = 65537                  # Стандартная открытая экспонента (число Ферма F4).
                                # Популярна: простая, обеспечивает быстрое возведение
                                # в степень и, как правило, взаимно проста с φ(n)
    while math.gcd(e, phi) != 1:       # Обязательное условие: НОД(e, φ(n)) = 1,
                                        # иначе обратный элемент не существует
        e = random.randrange(2, phi)    # Если условие нарушено — выбираем случайное e

    d = mod_inverse(e, phi)    # Вычисляем закрытую экспоненту d: обратную к e по модулю φ(n)
    return (e, n), (d, n)      # Возвращаем пару: открытый ключ (e, n) и закрытый ключ (d, n)
```

### Функция `encrypt()` — Шифрование

```python
def encrypt(message: int, public_key) -> int:
    e, n = public_key            # Распаковываем открытый ключ на экспоненту e и модуль n
    return pow(message, e, n)   # Вычисляем c = m^e mod n.
                                  # Встроенная pow(base, exp, mod) использует быстрое
                                  # возведение в степень — эффективнее, чем (m**e) % n
```

---

### Функция `decrypt()` — Дешифрование

```python
def decrypt(ciphertext: int, private_key) -> int:
    d, n = private_key               # Распаковываем закрытый ключ на экспоненту d и модуль n
    return pow(ciphertext, d, n)    # Вычисляем m = c^d mod n.
                                      # Математически гарантировано, что результат совпадёт
                                      # с исходным сообщением m
```

### Блок `if __name__ == "__main__"` — Демонстрация работы

```python
if __name__ == "__main__":                          # Выполняется только при прямом запуске файла,
                                                     # но не при импорте как модуля
    public_key, private_key = generate_keys()       # Генерируем пару ключей RSA

    print(f"Public key:  (e={public_key[0]}, n={public_key[1]})")   # Выводим открытый ключ
    print(f"Private key: (d={private_key[0]}, n={private_key[1]})") # Выводим закрытый ключ

    message = 42                                    # Исходное сообщение в виде числа.
                                                     # Обязательное условие: message < n
    cipher = encrypt(message, public_key)           # Шифруем сообщение открытым ключом
    decoded = decrypt(cipher, private_key)          # Дешифруем результат закрытым ключом

    print(f"\nOriginal:  {message}")                # Исходное сообщение: 42
    print(f"Encrypted: {cipher}")                   # Зашифрованное число
    print(f"Decrypted: {decoded}")                  # Должно совпасть с исходным: 42
```

### Функция `find_top_department()` — Поиск отдела с наибольшим фондом зарплат

```python
def find_top_department(filepath="employees.json"):  # Путь к файлу задан по умолчанию
    with open(filepath, encoding="utf-8") as f:      # Открываем файл с явной кодировкой UTF-8,
                                                       # чтобы корректно читать кириллицу.
                                                       # with-блок автоматически закрывает файл
        employees = json.load(f)                      # Десериализуем JSON в список словарей Python.
                                                       # Каждый элемент — словарь с ключами:
                                                       # "name", "department", "salary"

    totals = defaultdict(int)          # Создаём словарь с дефолтным значением 0 (int()).
                                        # При обращении к новому ключу автоматически ставится 0,
                                        # поэтому можно сразу прибавлять без инициализации
    for emp in employees:              # Перебираем всех сотрудников
        totals[emp["department"]] += emp["salary"]   # Суммируем зарплаты по отделам.
                                                       # Ключ — название отдела, значение — сумма

    top_dept = max(totals, key=totals.get)  # Находим ключ (отдел) с максимальным значением.
                                              # key=totals.get означает: сравниваем не сами ключи,
                                              # а значения словаря totals для каждого ключа

    print(f"Отдел с наибольшей суммарной зарплатой: {top_dept} ({totals[top_dept]:,})")
    # Выводим название отдела и суммарную зарплату.
    # Формат {:,} добавляет разделитель тысяч: например, 150000 → 150,000

    print("\nСотрудники:")              # Заголовок списка сотрудников
    for emp in employees:              # Снова перебираем всех сотрудников
        if emp["department"] == top_dept:           # Фильтруем: только из топ-отдела
            print(f"  {emp['name']} — {emp['salary']:,}")  # Выводим имя и зарплату
                                                             # с разделителем тысяч
```
